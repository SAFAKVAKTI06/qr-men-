  <script>
    // GOOGLE SHEETS CSV LINKI BURAYA EKLENECEK
    const sheetCsvUrl = 'YOUR_GOOGLE_SHEETS_CSV_URL_HERE';

    let config = {
      fiyatLariGoster: true,
      garsonCagirAktif: true,
      whatsappNo: '',
      googleYorum: '',
      instagram: ''
    };

    let products = [];

    // Verileri Yükle
    async function init() {
      // Örnek Demolar (Google Sheets Baglanana Kadar)
      products = [
        { kategori: 'Dürüm', urun_adi: 'Yaprak Et Döner Dürüm', aciklama: 'Özel sos, amasya et, turşu, domates', fiyat: '180 ₺', gorsel_url: 'https://images.unsplash.com/photo-1561651823-34feb02250e4', kalori: '650 kcal', etiket: 'En Çok Satan', alerjen: 'Gluten' },
        { kategori: 'Porsiyon', urun_adi: 'İskender Döner', aciklama: 'Tereyağlı pide, özel sos, manda yoğurdu', fiyat: '260 ₺', gorsel_url: 'https://images.unsplash.com/photo-1529006557810-274b9b2fc783', kalori: '850 kcal', etiket: 'Şefin İmzası', alerjen: 'Süt, Gluten' }
      ];

      renderCategories();
      renderMenu(products);
      renderBottomBar();
    }

    function renderCategories() {
      const cats = ['Tümü', ...new Set(products.map(p => p.kategori))];
      document.getElementById('categoryContainer').innerHTML = cats.map((c, i) => `
        <button class="cat-btn ${i===0?'active':''}" onclick="filterCat('${c}', this)">${c}</button>
      `).join('');
    }

    function renderMenu(list) {
      const container = document.getElementById('menuContainer');
      container.innerHTML = list.map(p => `
        <div class="product-card" onclick="openModal('${p.urun_adi}')">
          <img src="${p.gorsel_url}" class="product-img">
          <div class="product-info">
            <div>
              <div class="product-title">${p.urun_adi}</div>
              <div class="product-desc">${p.aciklama}</div>
            </div>
            <div class="product-meta">
              <div class="price ${config.fiyatLariGoster ? '' : 'price-hidden'}">${p.fiyat}</div>
              <div class="badges">
                ${p.kalori ? `<span class="badge-item"><i class="fa-solid fa-fire"></i> ${p.kalori}</span>` : ''}
              </div>
            </div>
          </div>
        </div>
      `).join('');
    }

    function filterCat(cat, btn) {
      document.querySelectorAll('.cat-btn').forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      if(cat === 'Tümü') renderMenu(products);
      else renderMenu(products.filter(p => p.kategori === cat));
    }

    function openModal(name) {
      const p = products.find(item => item.urun_adi === name);
      if(!p) return;
      document.getElementById('modalImg').src = p.gorsel_url;
      document.getElementById('modalTitle').innerText = p.urun_adi;
      document.getElementById('modalDesc').innerText = p.aciklama;
      document.getElementById('modalMeta').innerHTML = `
        <p><strong>Kalori:</strong> ${p.kalori || 'Belirtilmedi'}</p>
        <p style="margin-top:5px;"><strong>Alerjenler:</strong> ${p.alerjen || 'Yok'}</p>
        <h3 style="color:var(--primary-orange); margin-top:15px; ${config.fiyatLariGoster ? '' : 'display:none;'}">${p.fiyat}</h3>
      `;
      document.getElementById('productModal').style.display = 'flex';
    }

    function closeModal() { document.getElementById('productModal').style.display = 'none'; }

    function renderBottomBar() {
      document.getElementById('bottomBar').innerHTML = `
        <a href="#" class="action-btn"><i class="fa-solid fa-location-dot"></i>Yol Tarifi</a>
        ${config.garsonCagirAktif ? `<a href="https://wa.me/${config.whatsappNo}?text=Masa%20Hesap%20/ %20Garson%20Istiyor" class="action-btn waiter"><i class="fa-solid fa-bell"></i>Garson Çağır</a>` : ''}
        <a href="#" class="action-btn"><i class="fa-solid fa-star"></i>Yorum Yap</a>
      `;
    }

    init();
  </script>
